Advanced JavaScript

📖 Reading time: 30 minutes 🎯 Difficulty: Advanced

ES6+ Features

ES6 (ECMAScript 2015) introduced many powerful features. Here are the most important ones:

📄 Template Literals
let name = "John";
let age = 25;

// Old way (string concatenation)
let old = "My name is " + name + " and I am " + age + " years old.";

// New way (template literals)
let modern = `My name is ${name} and I am ${age} years old.`;

// Multi-line strings
let html = `
    <div class="card">
        <h2>${name}</h2>
        <p>Age: ${age}</p>
    </div>
`;
📄 Destructuring
// Array destructuring
let [a, b, c] = [1, 2, 3];
console.log(a);  // 1
console.log(b);  // 2

// Object destructuring
let person = { name: "John", age: 25, city: "NYC" };
let { name, age, city } = person;
console.log(name);  // "John"

// Skip properties
let { name: userName, age: userAge } = person;

// Default values
let { name, age, job = "developer" } = person;
📄 Spread & Rest Operators
// Spread operator (...)
let arr1 = [1, 2, 3];
let arr2 = [...arr1, 4, 5];  // [1, 2, 3, 4, 5]

let obj1 = { a: 1, b: 2 };
let obj2 = { ...obj1, c: 3 };  // { a: 1, b: 2, c: 3 }

// Copy array
let copy = [...arr1];

// Rest parameters
function sum(...numbers) {
    return numbers.reduce((total, num) => total + num, 0);
}
console.log(sum(1, 2, 3, 4));  // 10
📄 Optional Chaining & Nullish Coalescing
// Optional chaining (?.)
let user = {
    name: "John",
    address: {
        city: "NYC"
    }
};

console.log(user?.address?.city);     // "NYC"
console.log(user?.phone?.number);     // undefined (no error!)

// Nullish coalescing (??)
let value = null;
let defaultVal = value ?? "default";  // "default"

let zero = 0;
let result = zero ?? "default";  // 0 (only null/undefined)

Classes

Classes are templates for creating objects. They're like blueprints for houses - you build the blueprint once, then create many houses from it.

📄 Class Basics
// Creating a class
class Person {
    // Constructor - runs when you create a new object
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }
    
    // Method
    greet() {
        return `Hello, I'm ${this.name} and I'm ${this.age} years old.`;
    }
    
    // Getter
    get info() {
        return `${this.name} (${this.age})`;
    }
    
    // Static method (called on class, not instance)
    static create(name, age) {
        return new Person(name, age);
    }
}

// Creating objects from the class
let john = new Person("John", 25);
let jane = new Person("Jane", 30);

console.log(john.greet());  // "Hello, I'm John and I'm 25 years old."
console.log(jane.info);     // "Jane (30)"

// Static method
let bob = Person.create("Bob", 35);
📄 Inheritance
// Parent class
class Animal {
    constructor(name) {
        this.name = name;
    }
    
    speak() {
        return `${this.name} makes a sound.`;
    }
}

// Child class (inherits from Animal)
class Dog extends Animal {
    constructor(name, breed) {
        super(name);  // Call parent constructor
        this.breed = breed;
    }
    
    // Override parent method
    speak() {
        return `${this.name} barks!`;
    }
    
    // New method
    fetch() {
        return `${this.name} fetches the ball.`;
    }
}

let rex = new Dog("Rex", "German Shepherd");
console.log(rex.speak());   // "Rex barks!"
console.log(rex.fetch());   // "Rex fetches the ball."

Asynchronous JavaScript

JavaScript runs code line by line. But some operations take time (like fetching data). Async programming lets JavaScript do other things while waiting.

💡 Think of it this way:

Imagine you're at a restaurant. You order food (async operation). You don't just stand there staring at the kitchen - you chat with friends (other code runs) until your food arrives (operation completes).

📄 Callbacks
// Callback function
function fetchData(callback) {
    setTimeout(() => {
        let data = { name: "John", age: 25 };
        callback(data);
    }, 1000);
}

fetchData(function(data) {
    console.log(data);  // After 1 second
});

// Problem: Callback Hell!
fetchData(function(data1) {
    fetchData(data1.id, function(data2) {
        fetchData(data2.id, function(data3) {
            // Hard to read and maintain!
        });
    });
});
📄 Promises
// Promise - represents a future value
function fetchData() {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            let success = true;
            if (success) {
                resolve({ name: "John", age: 25 });
            } else {
                reject("Error fetching data");
            }
        }, 1000);
    });
}

// Using promises
fetchData()
    .then(data => console.log(data))
    .catch(error => console.log(error))
    .finally(() => console.log("Done!"));

// Chaining promises
fetchData()
    .then(data => process(data))
    .then(processed => save(processed))
    .then(() => console.log("Saved!"))
    .catch(err => console.log(err));
📄 Async/Await (Modern!)
// Async/Await - cleaner way to handle promises
async function getData() {
    try {
        let data = await fetchData();
        console.log(data);
        
        let processed = await processData(data);
        console.log(processed);
        
        await saveData(processed);
        console.log("Saved!");
    } catch (error) {
        console.log("Error:", error);
    }
}

// Call the async function
getData();

Fetch API

The Fetch API lets you make HTTP requests to servers (like getting data from an API).

📄 Fetch API
// Basic GET request
fetch("https://api.example.com/users")
    .then(response => response.json())
    .then(data => console.log(data))
    .catch(error => console.log(error));

// Async/Await version
async function getUsers() {
    try {
        let response = await fetch("https://api.example.com/users");
        let data = await response.json();
        console.log(data);
    } catch (error) {
        console.log("Error:", error);
    }
}

// POST request
async function createUser(user) {
    let response = await fetch("https://api.example.com/users", {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify(user)
    });
    return await response.json();
}

Local Storage

Local Storage lets you save data in the browser. The data persists even when the page is refreshed.

📄 Local Storage
// Save data
localStorage.setItem("name", "John");
localStorage.setItem("age", "25");

// Save objects (convert to JSON)
let user = { name: "John", age: 25 };
localStorage.setItem("user", JSON.stringify(user));

// Get data
let name = localStorage.getItem("name");  // "John"
let user = JSON.parse(localStorage.getItem("user"));

// Remove data
localStorage.removeItem("name");

// Clear all data
localStorage.clear();

// Check if key exists
let exists = localStorage.getItem("name") !== null;
✅ Use Cases:
  • Save user preferences (theme, language)
  • Cache data from API
  • Save form data (prevent data loss)
  • Build offline-capable apps

Error Handling

📄 Try/Catch
// Try/Catch block
try {
    // Code that might cause an error
    let result = riskyOperation();
    console.log(result);
} catch (error) {
    // Handle the error
    console.log("Error:", error.message);
} finally {
    // Always runs (whether there's an error or not)
    console.log("Cleanup code");
}

// Throwing custom errors
function divide(a, b) {
    if (b === 0) {
        throw new Error("Cannot divide by zero!");
    }
    return a / b;
}

try {
    console.log(divide(10, 0));
} catch (error) {
    console.log(error.message);  // "Cannot divide by zero!"
}
